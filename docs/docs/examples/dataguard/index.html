Data Guard

Primary and standby database with Data Guard on 2 Oracle Linux VMs.

The example is described step by step in the guide.

Source: example/dataguard

# RMAN backups with NFS (playbooks/nfs_server.yml, role oradb_rman)
#
# dataguard-dbfs1 (group nfs_server) exports nfs_server_backup_path read write,
# the hosts of the group nfs_client mount the share permanently on the same path.
# => all members of the Data Guard configuration write their backups into the same directory
#    and every member can read the backups of the other member.

nfs_server_backup_path: /u01/app/oracle/backup
# Mount options of the backup share on the NFS clients
nfs_client_backup_opts: rw,bg,hard,tcp,timeo=600,rsize=1048576,wsize=1048576

# CONFIGURE DEVICE TYPE DISK in the RMAN script parameter (role oradb_rman)
rman_device_type_disk: 'PARALLELISM 1 BACKUP TYPE TO COMPRESSED BACKUPSET'

# Database DB1 on both members of the Data Guard configuration DG1.
#
# Host specific values in host_vars/<host>/databases.yml:
# - db1_unique_name: db_unique_name of the member in db_standby_config (group_vars/dbfs/standby.yml)
#
# dataguard-dbfs1 (initial_role primary):          created with dbca
# dataguard-dbfs2 (initial_role physical_standby): created with RMAN DUPLICATE by oradb_standby
oracle_database_db1:
  home: &db_config_home db26_si_ee
  oracle_db_name: &oracle_db_name DB1
  oracle_db_unique_name: "{{ db1_unique_name }}"
  standby_config: DG1
  oracle_db_type: SI                                   # Type of database (RAC,RACONENODE,SI)
  storage_type: FS                                     # Database storage to be used. ASM or FS.
  oracle_database_type: MULTIPURPOSE                   # MULTIPURPOSE|DATA_WAREHOUSING|OLTP
  redolog_size: 75M
  redolog_groups: 3
  # mandatory for Data Guard: static listener entry <db_unique_name>_DGMGRL
  listener_name: LISTENER
  listener_port: &cdb_listener_port 1521
  # Data Guard needs archivelog and force logging, flashback allows the reinstate after a failover
  archivelog: true
  flashback: true
  force_logging: true
  state: present
  tablespaces:
    - {name: system, size: 10M, autoextend: true, next: 50M, maxsize: 4G, content: permanent, state: present, bigfile: false}
    - {name: sysaux, size: 10M, autoextend: true, next: 50M, maxsize: 4G, content: permanent, state: present, bigfile: false}

    - {name: users, size: 10M, autoextend: true, next: 50M, maxsize: 2G, content: permanent, state: present, bigfile: false}
    - {name: temp, size: 10M, autoextend: true, next: 50M, maxsize: 4G, content: permanent, state: present, bigfile: false}
  init_parameters:
    - {name: processes, value: '400', scope: spfile, state: present, dbca: false}
    # EZConnect of the local listener with the IP address of the host.
    # The standby gets its own value with SET in RMAN DUPLICATE (oradb_standby).
    - {name: local_listener, value: '//{{ ansible_host }}:1521', scope: both, state: present}
    - {name: control_management_pack_access, value: 'NONE', scope: both, state: present}
    - {name: control_file_record_keep_time, value: '30', scope: both, state: present}
    - {name: deferred_segment_creation, value: 'false', scope: both, state: present}
    - {name: filesystemio_options, value: 'setall', scope: spfile, state: present}
    - {name: job_queue_processes, value: '10', scope: both, state: present}

    - {name: sga_target, value: '1800M', scope: spfile, state: present}
    - {name: recyclebin, value: 'off', scope: spfile, state: present}
    - {name: max_pdbs, value: '3', scope: both, state: present}
    # Oracle Managed Files are mandatory for Data Guard in ansible-oracle
    - {name: db_create_file_dest, value: '/u02/oradata', scope: both, state: present}
    - {name: db_create_online_log_dest_1, value: '/u02/oradata', scope: both, state: present}
    - {name: db_recovery_file_dest_size, value: '10G', scope: both, state: present, dbca: false}
    - {name: db_recovery_file_dest, value: '/u02/fra', scope: both, state: present}
    # Data Guard Broker: configuration files outside of ORACLE_HOME (read-only ORACLE_HOME => ORACLE_BASE/dbs),
    # dg_broker_start below the file parameters: the files cannot be changed while the broker is running
    - {name: dg_broker_config_file1, value: '/u01/app/oracle/dbs/dr1{{ db1_unique_name }}.dat', scope: both, state: present, dbca: false}

  profiles:
    - name: DEFAULT
      state: present
      attributes:
        - {name: password_life_time, value: unlimited}

  # RMAN backups to the NFS share nfs_server_backup_path (group_vars/all/backup.yml)

ansible/inventory/group_vars/dbfs/standby.yml

---
# Data Guard configuration DG1 with 2 members.
# Must be identical on all members => group_vars of the group with all members.
#
# Inventory structure: doc/guides/standby_databases.adoc
db_standby_config:
  - name: DG1
    type: dataguard
    dg:
      protection_mode: MaxPerformance
      # mandatory: RMAN CONFIGURE ARCHIVELOG DELETION POLICY of all members (oradb_standby).
      # Archive logs are eligible for deletion after they are applied on all standby databases.
      # 'BACKED UP ...' needs an RMAN catalog: without catalog the controlfile of a standby
      # does not know the backups of the primary => no archive log of the standby would be eligible.
      archivelog_deletion_policy: APPLIED ON ALL STANDBY
    members:
      - hosts: [dataguard-dbfs1]
        db_unique_name: DB1_SITE1
        initial_role: primary
        dg:
          connect_identifier: DB1_SITE1
          log_xpt_mode: ASYNC
      - hosts: [dataguard-dbfs2]
        db_unique_name: DB1_SITE2
        initial_role: physical_standby
        dg:
          connect_identifier: DB1_SITE2
          log_xpt_mode: ASYNC

ansible/playbooks/autoupgrade.yml

---
# AutoUpgrade on the NFS server only (group nfs_server):
# the gold image is downloaded once into /u01/sw and used by all database servers with NFS.
- name: Install Oracle AutoUpgrade
  hosts: "{{ hostgroup | default('nfs_server') }}"
  any_errors_fatal: true
  tasks:

    - name: install openJDK and python-pexpect
      become: true
      become_user: root
      when:
        - not autoupgrade_run_local | bool
      ansible.builtin.package:
        name:
          - java-1.8.0-openjdk
          - python-pexpect
        state: present

    - name: Install autoupgrade.jar
      ansible.builtin.import_role:
        name: opitzconsulting.ansible_oracle.autoupgrade
        tasks_from: install.yml

- name: Import_playbook autoupgrade_download_patches
  import_playbook: opitzconsulting.ansible_oracle.autoupgrade_download_patches
  vars:
    hostgroup: nfs_server

ansible/playbooks/nfs_server.yml

---
# NFS server on the 1st database server (group nfs_server) for
# - the installation media (nfs_server_sw_path, read only):
#   AutoUpgrade downloads the gold image into nfs_server_sw_path on this host only.
#   The hosts of the group nfs_client mount the share with install_from_nfs: true
#   during the installation of the ORACLE_HOME (role oraswdb_install).
# - the RMAN backups (nfs_server_backup_path, read write):
#   The hosts of the group nfs_client mount the share permanently on the same path (2nd play).
#
# Requires the playbook opitzconsulting.ansible_oracle.os before: filesystem /u01, user oracle.
- name: NFS server for installation media and RMAN backups
  hosts: "{{ hostgroup | default('nfs_server') }}"
  become: true
  any_errors_fatal: true
  vars:
    # IP addresses of the NFS clients
    nfs_server_clients: >-
      {{ groups['nfs_client'] | default([]) | map('extract', hostvars, 'ansible_host') | list }}
  tasks:
    - name: Install nfs-utils
      ansible.builtin.package:
        name: nfs-utils
        state: present

    # target of the AutoUpgrade download
    - name: Create directory for installation media
      ansible.builtin.file:
        path: "{{ nfs_server_sw_path }}"
        state: directory
        owner: "{{ oracle_user | default('oracle') }}"
        group: "{{ oracle_group | default('oinstall') }}"
        mode: "0755"

    - name: Export installation media
      ansible.builtin.copy:
        content: |
          # Managed by Ansible (playbooks/nfs_server.yml): installation media for ansible-oracle
          {{ nfs_server_sw_path }} {{ nfs_server_clients | map('regex_replace', '$', '(ro,sync,root_squash)') | join(' ') }}
        dest: /etc/exports.d/oracle-sw.exports
        owner: root
        group: root
        mode: "0644"
      register: _nfs_server_exports

    # ORACLE_BASE with the same owner and mode as in the role oraswdb_install
    - name: Create ORACLE_BASE
      ansible.builtin.file:
        path: "{{ oracle_base | default('/u01/app/oracle') }}"
        state: directory
        owner: "{{ oracle_user | default('oracle') }}"
        group: "{{ oracle_group | default('oinstall') }}"
        mode: "0775"

    # RMAN does not create directories: one directory per DB_NAME (%d in the RMAN format)
    - name: Create directories for RMAN backups
      ansible.builtin.file:
        path: "{{ item }}"
        state: directory
        owner: "{{ oracle_user | default('oracle') }}"
        group: "{{ oracle_group | default('oinstall') }}"
        mode: "0750"
      loop: >-
        {{ [nfs_server_backup_path]
           + oracle_databases | default([]) | map(attribute='oracle_db_name')
             | map('regex_replace', '^', nfs_server_backup_path ~ '/') | list }}

    - name: Export RMAN backups
      ansible.builtin.copy:
        content: |
          # Managed by Ansible (playbooks/nfs_server.yml): RMAN backups for ansible-oracle
          {{ nfs_server_backup_path }} {{ nfs_server_clients | map('regex_replace', '$', '(rw,sync,root_squash)') | join(' ') }}
        dest: /etc/exports.d/oracle-backup.exports
        owner: root
        group: root
        mode: "0644"
      register: _nfs_server_exports_backup

    - name: Start NFS server
      ansible.builtin.systemd:
        name: nfs-server
        enabled: true
        state: started

    - name: Reload exports
      ansible.builtin.command: exportfs -ra
      when: _nfs_server_exports is changed or _nfs_server_exports_backup is changed  # noqa no-handler
      changed_when: true

    - name: Gather service facts
      ansible.builtin.service_facts:

    - name: Open NFS in firewalld
      ansible.posix.firewalld:
        service: "{{ item }}"
        permanent: true
        immediate: true
        state: enabled
      loop:
        - nfs
        - rpc-bind
        - mountd
      when:
        - ansible_facts.services['firewalld.service'] is defined
        - ansible_facts.services['firewalld.service'].state == 'running'

# Permanent mount of the RMAN backups on the NFS clients (/etc/fstab).
